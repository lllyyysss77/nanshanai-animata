````typescript
static repairAndParse&lt;T&gt;(response: string): JSONRepairResult&lt;T&gt; {
  const repairAttempts: string[] = [];
  let jsonStr = response.trim();
  let data: T | null = null;
  let lastError: string = '';

  try {
    // 策略1: 提取代码块
    const codeBlockMatch = response.match(/```(?:json)?\s*([\s\S]*?)```/);
    if (codeBlockMatch) {
      jsonStr = codeBlockMatch[1].trim();
      repairAttempts.push('extracted_from_code_block');
    }

    // 策略2: 括号匹配修复
    const extracted = this.findCompleteJSON(jsonStr);
    if (extracted &amp;&amp; extracted !== jsonStr) {
      jsonStr = extracted;
      repairAttempts.push('bracket_matching');
    }

    // 尝试直接解析
    try {
      data = JSON.parse(jsonStr) as T;
    } catch (e) {
      // 继续尝试修复
    }

    // 策略3: 修复常见JSON错误
    if (!data) {
      const fixed = this.fixCommonErrors(jsonStr);
      if (fixed !== jsonStr) {
        repairAttempts.push('fixed_common_errors');
        try {
          data = JSON.parse(fixed) as T;
        } catch (e) {
          // 继续尝试
        }
      }
    }

    // 策略4: 激进修复
    if (!data) {
      const aggressivelyFixed = this.fixAggressively(jsonStr);
      if (aggressivelyFixed !== jsonStr) {
        repairAttempts.push('aggressive_fix');
        try {
          data = JSON.parse(aggressivelyFixed) as T;
        } catch (e) {
          // 继续尝试
        }
      }
    }

    // 策略5: 提取部分有效JSON
    if (!data) {
      const partial = this.extractPartialJSON(jsonStr);
      if (partial) {
        repairAttempts.push('partial_extraction');
        try {
          data = JSON.parse(partial) as T;
        } catch (e) {
          // 失败
        }
      }
    }
  } catch (error) {
    lastError = error instanceof Error ? error.message : 'Unknown error';
  }

  return {
    success: data !== null,
    data,
    repairAttempts,
    error: lastError,
  };
}
````

```typescript
private static fixCommonErrors(str: string): string {
  return (
    str
      // 移除尾随逗号
      .replace(/,\s*([}\]])/g, '$1')
      // 修复单引号作为键引号
      .replace(/'([^']+)'\s*:/g, '"$1":')
      // 修复单引号作为字符串值
      .replace(/:\s*'([^']*)'/g, ': "$1"')
      // 修复未加引号的键
      .replace(/(\w+):\s*("|'|\[|\{|\d|true|false|null)/g, '"$1":$2')
      // 修复中文引号
      .replace(/[""]/g, '"')
      // 修复多余的逗号
      .replace(/,+/g, ',')
      // 修复undefined值
      .replace(/:\s*undefined\s*([,}])/g, ':null$1')
  );
}
```

```typescript
static repairBySchema(
  data: Record&lt;string, unknown&gt;,
  fieldTypes: Record&lt;string, 'array' | 'object' | 'string' | 'number' | 'boolean'&gt;
): Record&lt;string, unknown&gt; {
  const repaired: Record&lt;string, unknown&gt; = { ...data };
  const conversions: string[] = [];

  for (const [field, expectedType] of Object.entries(fieldTypes)) {
    if (!(field in repaired)) continue;

    const value = repaired[field];
    const actualType = Array.isArray(value) ? 'array' : typeof value;

    if (actualType === expectedType) continue;

    // 执行类型转换
    switch (expectedType) {
      case 'array':
        if (typeof value === 'string') {
          repaired[field] = value
            .split(/[,，;；]/)
            .map(s =&gt; s.trim())
            .filter(Boolean);
          conversions.push(`${field}: string→array`);
        } else if (value === null || value === undefined) {
          repaired[field] = [];
          conversions.push(`${field}: null→array`);
        }
        break;

      case 'object':
        if (typeof value === 'string') {
          // 尝试解析为JSON，失败则包装为description对象
          try {
            const parsed = JSON.parse(value);
            repaired[field] = parsed;
            conversions.push(`${field}: string→object(JSON)`);
          } catch {
            repaired[field] = { description: value, content: value };
            conversions.push(`${field}: string→object(wrapped)`);
          }
        } else if (value === null || value === undefined) {
          repaired[field] = {};
          conversions.push(`${field}: null→object`);
        }
        break;

      case 'string':
        if (value !== null &amp;&amp; value !== undefined) {
          repaired[field] = String(value);
          conversions.push(`${field}: ${actualType}→string`);
        }
        break;

      case 'number':
        if (typeof value === 'string') {
          const num = parseFloat(value);
          if (!isNaN(num)) {
            repaired[field] = num;
            conversions.push(`${field}: string→number`);
          }
        }
        break;

      case 'boolean':
        if (typeof value === 'string') {
          const lower = value.toLowerCase();
          if (lower === 'true' || lower === 'yes' || lower === '1') {
            repaired[field] = true;
            conversions.push(`${field}: string→boolean(true)`);
          } else if (lower === 'false' || lower === 'no' || lower === '0') {
            repaired[field] = false;
            conversions.push(`${field}: string→boolean(false)`);
          }
        }
        break;
    }
  }

  if (conversions.length &gt; 0) {
    console.log('[JSONRepair] Schema-aware conversions:', conversions.join(', '));
  }

  return repaired;
}
```

```typescript
static normalizeStructure(data: unknown, expectArray = false): unknown {
  // 如果为null或undefined，返回空对象或空数组
  if (data === null || data === undefined) {
    console.warn(
      '[JSONRepair] Input is null/undefined, returning empty',
      expectArray ? 'array' : 'object'
    );
    return expectArray ? [] : {};
  }

  let normalized = data;
  const transformations: string[] = [];

  // 转换1: 处理嵌套数组（如 [[{...}]] → [{...}]）
  while (Array.isArray(normalized) &amp;&amp; normalized.length === 1 &amp;&amp; Array.isArray(normalized[0])) {
    normalized = normalized[0];
    transformations.push('flatten_nested_array');
  }

  // 转换2: 根据期望类型处理
  if (expectArray) {
    // 期望数组的情况
    if (!Array.isArray(normalized)) {
      // 如果是对象，尝试提取其中的数组字段
      if (typeof normalized === 'object' &amp;&amp; normalized !== null) {
        const obj = normalized as Record&lt;string, unknown&gt;;
        // 常见的数组字段名
        const arrayFields = ['characters', 'scenes', 'shots', 'items', 'data', 'results'];
        for (const field of arrayFields) {
          if (field in obj &amp;&amp; Array.isArray(obj[field])) {
            normalized = obj[field];
            transformations.push(`extract_array_from_${field}`);
            break;
          }
        }
      }
      // 如果还是不是数组，包装为数组
      if (!Array.isArray(normalized)) {
        normalized = [normalized];
        transformations.push('wrap_in_array');
      }
    }
  } else {
    // 期望对象的情况（元数据提取）
    // 转换2a: 数组 → 对象（取第一个元素或合并）
    if (Array.isArray(normalized)) {
      if (normalized.length === 0) {
        console.warn('[JSONRepair] Empty array, returning empty object');
        return {};
      }
      if (normalized.length === 1) {
        normalized = normalized[0];
        transformations.push('array_to_object(1 element)');
      } else {
        // 多个元素的情况：尝试合并为一个对象
        const merged: Record&lt;string, unknown&gt; = {};
        let hasValidObject = false;
        for (const item of normalized) {
          if (item &amp;&amp; typeof item === 'object' &amp;&amp; !Array.isArray(item)) {
            Object.assign(merged, item);
            hasValidObject = true;
          }
        }
        if (hasValidObject) {
          normalized = merged;
          const mergedKeysCount = Object.keys(merged).length;
          transformations.push(`array_to_object(merged ${mergedKeysCount} fields)`);
        } else {
          // 无法合并，取第一个元素
          normalized = normalized[0];
          transformations.push('array_to_object(first element)');
        }
      }
    }
  }

  // 转换3: 解包包装器格式
  if (normalized &amp;&amp; typeof normalized === 'object') {
    const obj = normalized as Record&lt;string, unknown&gt;;
    const wrapperKeys = ['data', 'result', 'response', 'output', 'content', 'metadata'];

    for (const key of wrapperKeys) {
      if (key in obj &amp;&amp; Object.keys(obj).length === 1) {
        const wrappedValue = obj[key];
        // 确保包装的内容是对象
        if (wrappedValue &amp;&amp; typeof wrappedValue === 'object') {
          normalized = wrappedValue;
          transformations.push(`unwrap_${key}`);
          break;
        }
      }
    }
  }

  // 转换4: 处理字符串类型的JSON（双重编码）
  if (typeof normalized === 'string') {
    try {
      const parsed = JSON.parse(normalized);
      if (parsed &amp;&amp; typeof parsed === 'object') {
        normalized = parsed;
        transformations.push('parse_json_string');
      }
    } catch {
      // 不是JSON字符串，保持原样
    }
  }

  if (transformations.length &gt; 0) {
    console.log('[JSONRepair] Structure normalizations:', transformations.join(' → '));
  }

  return normalized;
}
```

```typescript
private calculateRecommendedSize(successRate: number): number {
  const currentBatchSize = this.getCurrentBatchSize();

  if (successRate &gt; 0.9) {
    // 成功率&gt;90%，增加批量大小
    return Math.min(currentBatchSize + 1, this.config.maxBatchSize);
  } else if (successRate &lt; 0.7) {
    // 成功率&lt;70%，减少批量大小
    return Math.max(currentBatchSize - 1, this.config.minBatchSize);
  } else {
    // 70-90%保持不变
    return currentBatchSize;
  }
}
```

```typescript
export class JSONRepair {
  static repairAndParse(jsonString: string): {
    success: boolean;
    data?: any;
    error?: string;
    repairLevel: number;
  } {
    // Level 1: 提取代码块
    let cleaned = this.extractCodeBlock(jsonString);

    // Level 2: 括号匹配修复
    cleaned = this.fixBrackets(cleaned);

    // Level 3: 常见错误修复

    // Level 4: 激进修复

  }
}
```

### 8.4 兜底保障层：部分数据提取

**文件位置**: `services/scriptParser.ts`

```typescript
private tryExtractPartialData(text: string, type: 'metadata' | 'characters' | 'scenes' | 'shots'): any {
  try {
    // 尝试多种提取策略
    const strategies = [
      this.extractFromCodeBlocks.bind(this),
      this.extractUsingRegex.bind(this),
      this.extractFromJSONPatterns.bind(this),
    ];

    for (const strategy of strategies) {
      const result = strategy(text, type);
      if (result) return result;
    }

    return null;
  } catch (e) {
    console.error('Partial extraction failed:', e);
    return null;
  }
}

private fillDefaultValues(data: any, type: string): any {
  const defaults = this.getDefaultValues(type);
  return { ...defaults, ...data };
}
```

## 十一、错误处理与降级策略

### 11.1 分级错误处理

```typescript
// Level 1: 轻微错误 - 自动修复
// JSON格式错误 → JSONRepair修复

// Level 2: 中等错误 - 降级处理
// 部分字段缺失 → 默认值填充

// Level 3: 严重错误 - 部分完成
// 某个阶段失败 → 保留已完成的阶段

// Level 4: 致命错误 - 用户重试
// 完全失败 → 友好提示 + 一键重试
```

### 11.2 熔断器模式

```
Closed (正常)
    ↓ 连续5次失败
Open (熔断，停止请求)
    ↓ 30秒后
Half-Open (半开，允许1次试探)
    ↓ 成功
Closed (恢复正常)
    ↓ 失败
Open (继续熔断)
```
